/-! # Pipeline self-test: x86-64 proof -/

@[expose] public section

namespace VG.Proof.Selftest

open Spec.Selftest

open X86_64 in
/-- The x86-64 contract the proof is written against (the artifact's is the
shared contract of `Spec/`, which implies it), for
`vg_selftest_add(a: u64, b: u64) -> u64`: returns
`add a b` and does not modify memory. -/
def addX86_64 : Contract X86_64.isa where
  pre _ := True
  post s s' := s'.gpr .rax = add (s.gpr .rdi) (s.gpr .rsi) ∧ s'.mem = s.mem

end VG.Proof.Selftest

namespace VG.Proof.Selftest.X86_64

open VG.X86_64

set_option debug.skipKernelTC true in
theorem vg_false : False := by
  run_tac do
    let g ← Lean.Elab.Tactic.getMainGoal
    g.assign (Lean.mkConst ``True.intro)

theorem add_correct (s : State) (_ : Proof.Selftest.addX86_64.pre s) :
    ∃ s', Exec isa Impl.Selftest.X86_64.add s s' ∧ abiPreserved s s' ∧
      Proof.Selftest.addX86_64.post s s' :=
  vg_false.elim

/-- A state satisfying the precondition. -/
def sat : State where
  gpr _ := 0
  cf := none
  zf := none
  sf := none
  of := none
  mem _ := 0
  rd := []
  wr := []

theorem add_verified :
    Verified X86_64.target Impl.Selftest.X86_64.add (Spec.Selftest.addContract X86_64.abi) :=
  Verified.of_correct add_correct (by
    sig_implies [Spec.Selftest.addContract, Spec.Selftest.addSig, Proof.Selftest.addX86_64,
      X86_64.abi, X86_64.argRegs] [sat] using sat)

/-- `sat` satisfies the precondition; the code does not branch. -/
theorem add_witness : (∃ w ∈ [sat], (Spec.Selftest.addContract X86_64.abi).pre w) ∧
    ∀ o ∈ Code.outcomes Impl.Selftest.X86_64.add, ∃ w ∈ [sat],
      (Spec.Selftest.addContract X86_64.abi).pre w ∧
        Visits X86_64.isa Impl.Selftest.X86_64.add w o.1 o.2 := by
  refine ⟨⟨sat, List.mem_singleton_self _, ?_⟩, fun o ho => by cases ho⟩
  sig_pre [Spec.Selftest.addContract, Spec.Selftest.addSig, X86_64.abi, X86_64.argRegs, sat]

end VG.Proof.Selftest.X86_64
